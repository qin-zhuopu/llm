    Alpine.data('newsCarousel', () => ({
        activeGroup: 0,
        isScrolling: false,
        totalItems: 0,
        visibleItems: 3,
        totalGroups: 0,
        
        init() {
            // Count the number of items in the carousel
            this.totalItems = this.$el.querySelectorAll('.carousel-item').length;
            
            // Adjust visible items based on screen size
            this.setVisibleItems();
            
            // Calculate total groups
            this.totalGroups = Math.ceil(this.totalItems / this.visibleItems);
            
            // Add resize listener to adjust visible items on window resize
            window.addEventListener('resize', () => this.setVisibleItems());
        },
        
        setVisibleItems() {
            // Set number of visible items based on screen width
            if (window.innerWidth < 640) {
                this.visibleItems = 1;
            } else {
                this.visibleItems = 3;
            }
            // Recalculate total groups
            this.totalGroups = Math.ceil(this.totalItems / this.visibleItems);
        },
        
        handleScroll() {
            if (this.isScrolling) return;
            
            const carousel = this.$refs.carousel;
            const scrollWidth = carousel.scrollWidth;
            const scrollLeft = carousel.scrollLeft;
            const clientWidth = carousel.clientWidth;
            
            // If we're near the end (within 100px of the end)
            if (scrollLeft + clientWidth >= scrollWidth - 100) {
                // Smoothly scroll back to the beginning after a short delay
                setTimeout(() => {
                    this.isScrolling = true;
                    carousel.scrollTo({
                        left: 0,
                        behavior: 'smooth'
                    });
                    
                    // Reset active group to 0
                    this.activeGroup = 0;
                    
                    // Reset isScrolling flag after animation completes
                    setTimeout(() => {
                        this.isScrolling = false;
                    }, 500);
                }, 300);
            } else {
                // Update active group based on scroll position
                const itemWidth = carousel.querySelector('.carousel-item').offsetWidth;
                const gap = 32; // 8px gap (adjust if your gap is different)
                const itemTotalWidth = itemWidth + gap;
                const currentPosition = scrollLeft;
                
                // Calculate which group is active
                this.activeGroup = Math.floor(currentPosition / (itemTotalWidth * this.visibleItems));
            }
        },
        
        goToGroup(groupIndex) {
            if (this.isScrolling) return;
            
            const carousel = this.$refs.carousel;
            const itemWidth = carousel.querySelector('.carousel-item').offsetWidth;
            const gap = 32; // 8px gap (adjust if your gap is different)
            const itemTotalWidth = itemWidth + gap;
            
            this.isScrolling = true;
            carousel.scrollTo({
                left: groupIndex * itemTotalWidth * this.visibleItems,
                behavior: 'smooth'
            });
            
            this.activeGroup = groupIndex;
            
            // Reset isScrolling flag after animation completes
            setTimeout(() => {
                this.isScrolling = false;
            }, 500);
        }
    }));
