<script>
(() => {
  "use strict";

  const initCoCoVideo = () => {
    const videoFrame = document.querySelector("[data-video-frame]");
    const video = videoFrame?.querySelector("[data-scroll-video]");

    if (!videoFrame || !video) {
      console.warn("CoCo hero video could not initialize.");
      return;
    }

    const variants = {
      landscape: {
        src: videoFrame.dataset.videoLandscape,
        width: 1920,
        height: 1080,
        ratio: 1920 / 1080
      },

      square: {
        src: videoFrame.dataset.videoSquare,
        width: 1080,
        height: 1080,
        ratio: 1
      },

      portrait: {
        src: videoFrame.dataset.videoPortrait,
        width: 1080,
        height: 1920,
        ratio: 1080 / 1920
      }
    };

    let activeVariant = "";
    let resizeTimer = null;

    const reducedMotionQuery = window.matchMedia(
      "(prefers-reduced-motion: reduce)"
    );

    const getClosestVariant = () => {
      const viewportRatio =
        Math.max(window.innerWidth, 1) /
        Math.max(window.innerHeight, 1);

      return Object.entries(variants).reduce(
        (closest, [name, variant]) => {
          const difference = Math.abs(
            viewportRatio - variant.ratio
          );

          return difference < closest.difference
            ? {
                name,
                difference
              }
            : closest;
        },
        {
          name: "landscape",
          difference: Math.abs(
            viewportRatio - variants.landscape.ratio
          )
        }
      ).name;
    };

    const showFallback = () => {
      videoFrame.classList.add("has-video-error");
      video.classList.remove("is-ready");
    };

    const showReducedMotionFallback = () => {
      video.pause();

      video.autoplay = false;
      video.removeAttribute("autoplay");

      video.removeAttribute("src");
      video.load();

      video.classList.remove("is-ready");
    };

    const playVideo = () => {
      if (reducedMotionQuery.matches) {
        showReducedMotionFallback();
        return;
      }

      video.muted = true;
      video.defaultMuted = true;

      video.play().catch((error) => {
        console.warn(
          "CoCo hero autoplay was blocked.",
          error
        );

        showFallback();
      });
    };

    const applyVariant = () => {
      const nextVariantName = getClosestVariant();
      const nextVariant = variants[nextVariantName];
      const fallbackImage = videoFrame.querySelector(
        ".coco-video-fallback"
      );
      
      const posterSrc =
        nextVariantName === "portrait"
          ? videoFrame.dataset.posterPortrait
          : nextVariantName === "square"
            ? videoFrame.dataset.posterSquare
            : videoFrame.dataset.posterLandscape;
      
      if (fallbackImage && posterSrc) {
        fallbackImage.classList.remove("is-poster-ready");
      
        fallbackImage.removeAttribute("srcset");
        fallbackImage.removeAttribute("sizes");
      
        const revealPoster = () => {
          fallbackImage.classList.add("is-poster-ready");
        };
      
        if (fallbackImage.src === posterSrc) {
          if (fallbackImage.complete) {
            revealPoster();
          } else {
            fallbackImage.addEventListener(
              "load",
              revealPoster,
              { once: true }
            );
          }
        } else {
          fallbackImage.addEventListener(
            "load",
            revealPoster,
            { once: true }
          );
      
          fallbackImage.src = posterSrc;
        }
      }

      if (!nextVariant?.src) {
        console.warn(
          `Missing CoCo video source for ${nextVariantName}.`
        );

        showFallback();
        return;
      }

      videoFrame.dataset.activeVideoVariant =
        nextVariantName;

      videoFrame.style.setProperty(
        "--video-author-width",
        `${nextVariant.width}px`
      );

      videoFrame.style.setProperty(
        "--video-author-height",
        `${nextVariant.height}px`
      );

      /*
       * Reduced motion:
       * keep the fallback image visible and
       * don't load or autoplay the video.
       */
      if (reducedMotionQuery.matches) {
        activeVariant = nextVariantName;
        showReducedMotionFallback();
        return;
      }

      const currentSource =
        video.getAttribute("src") || "";

      /*
       * If the selected responsive variant uses
       * the same source that's already loaded,
       * don't reload the video.
       */
      if (currentSource === nextVariant.src) {
        activeVariant = nextVariantName;
        playVideo();
        return;
      }

      const previousTime =
        Number.isFinite(video.currentTime) &&
        video.currentTime > 0
          ? video.currentTime
          : 0;

      activeVariant = nextVariantName;

      videoFrame.classList.remove("has-video-error");
      video.classList.remove("is-ready");

      video.pause();
      video.removeAttribute("src");
      video.load();

      video.setAttribute("src", nextVariant.src);
      video.load();

      video.addEventListener(
        "loadedmetadata",
        () => {
          if (
            previousTime > 0 &&
            Number.isFinite(video.duration)
          ) {
            const safeTime = Math.min(
              previousTime,
              Math.max(
                video.duration - 0.1,
                0
              )
            );

            try {
              video.currentTime = safeTime;
            } catch (error) {
              console.warn(
                "CoCo video playback position could not be restored.",
                error
              );
            }
          }

          playVideo();
        },
        { once: true }
      );
    };

    /*
     * Base video configuration
     */
    video.muted = true;
    video.defaultMuted = true;
    video.playsInline = true;

    video.setAttribute("muted", "");
    video.setAttribute("playsinline", "");

    /*
     * Only enable autoplay when the user
     * has not requested reduced motion.
     */
    if (!reducedMotionQuery.matches) {
      video.autoplay = true;
      video.setAttribute("autoplay", "");
    } else {
      video.autoplay = false;
      video.removeAttribute("autoplay");
    }

    /*
     * Reveal the video only after playback
     * has actually started.
     */
    video.addEventListener("playing", () => {
      videoFrame.classList.remove(
        "has-video-error"
      );

      video.classList.add("is-ready");
    });

    /*
     * Show fallback if the video fails.
     */
    video.addEventListener(
      "error",
      showFallback
    );

    /*
     * Initial setup
     */
    applyVariant();

    /*
     * Switch responsive video variant
     * when the viewport changes.
     */
    window.addEventListener(
      "resize",
      () => {
        window.clearTimeout(resizeTimer);

        resizeTimer = window.setTimeout(
          () => {
            applyVariant();
          },
          250
        );
      },
      { passive: true }
    );

    /*
     * Respond if the user changes their
     * OS/browser reduced-motion preference
     * while the page is already open.
     */
    reducedMotionQuery.addEventListener(
      "change",
      () => {
        if (reducedMotionQuery.matches) {
          showReducedMotionFallback();
        } else {
          video.autoplay = true;
          video.setAttribute(
            "autoplay",
            ""
          );

          applyVariant();
        }
      }
    );
  };

  if (document.readyState === "loading") {
    document.addEventListener(
      "DOMContentLoaded",
      initCoCoVideo,
      { once: true }
    );
  } else {
    initCoCoVideo();
  }
})();
</script>

<script>
(() => {
  "use strict";

  const initCoCoVideoSlider = () => {
    const slider = document.querySelector(
      "[data-coco-video-slider]"
    );

    const modal = document.querySelector(
      "[data-video-modal]"
    );

    const modalVideo = document.querySelector(
      "[data-modal-video]"
    );

    const modalYoutube = document.querySelector(
      "[data-modal-youtube]"
    );

    const closeButton = document.querySelector(
      "[data-video-close]"
    );

    if (!slider || typeof Swiper === "undefined") {
      console.warn(
        "CoCo video slider could not initialize."
      );
      return;
    }

    /*
     * REAL SLIDES
     */
    const realSlides = Array.from(
      slider.querySelectorAll(".swiper-slide")
    );

    const realSlideCount = realSlides.length;

    /*
     * ACCESSIBLE SLIDE LABELS
     */
    const setSlideLabels = (swiperInstance) => {
      swiperInstance.slides.forEach((slide, index) => {
        const trigger = slide.querySelector(
          "[data-video-open]"
        );

        const existingLabel =
          trigger?.getAttribute("aria-label");

        if (!existingLabel) return;

        const cleanLabel =
          existingLabel.replace(/^Play\s+/i, "");

        slide.setAttribute(
          "aria-label",
          `${cleanLabel}, slide ${index + 1} of ${swiperInstance.slides.length}`
        );
      });
    };

    /*
     * SWIPER
     */
    const swiper = new Swiper(slider, {
      slidesPerView: 1,
      slidesPerGroup: 1,
      centeredSlides: false,
      spaceBetween: 16,

      /*
       * IMPORTANT:
       * No loop clones. This prevents keyboard
       * focus from cycling through cloned slides.
       */
      loop: false,

      /*
       * Allows arrow/swipe navigation to wrap
       * without creating cloned slides.
       */
      rewind: true,

      speed: 600,
      watchOverflow: true,

      observer: true,
      observeParents: true,
      observeSlideChildren: true,
      resizeObserver: true,

      keyboard: {
        enabled: true
      },

      a11y: {
        enabled: true,
        prevSlideMessage: "Previous video",
        nextSlideMessage: "Next video",
        firstSlideMessage: "This is the first video",
        lastSlideMessage: "This is the last video"
      },

      on: {
        afterInit(swiperInstance) {
          setSlideLabels(swiperInstance);
        }
      },

      breakpoints: {
        480: {
          slidesPerView: 1,
          slidesPerGroup: 1,
          spaceBetween: 16
        },

        768: {
          slidesPerView: 1.2,
          slidesPerGroup: 1,
          spaceBetween: 20
        },

        992: {
          slidesPerView: 2,
          slidesPerGroup: 1,
          spaceBetween: 20
        }
      }
    });

    /*
     * CUSTOM PAGINATION
     * One dot per REAL video.
     *
     * selectedIndex is intentionally separate
     * from swiper.activeIndex.
     *
     * Example with 2 slides visible:
     *
     * Dot 3 -> [3][4]
     * Dot 4 -> [3][4]
     *
     * Swiper cannot move #4 farther left, but
     * dot 4 can still correctly remain selected.
     */
    const pagination = slider.querySelector(
      ".swiper-pagination"
    );

    if (pagination && realSlideCount) {
      pagination.innerHTML = "";

      const bullets = [];
      let selectedIndex = 0;

      const updatePagination = (
        requestedIndex = null
      ) => {
        if (requestedIndex !== null) {
          selectedIndex = requestedIndex;
        } else {
          selectedIndex = swiper.activeIndex;
        }

        bullets.forEach((bullet, index) => {
          const active =
            index === selectedIndex;

          bullet.classList.toggle(
            "swiper-pagination-bullet-active",
            active
          );

          if (active) {
            bullet.setAttribute(
              "aria-current",
              "true"
            );

            bullet.setAttribute(
              "aria-label",
              `Current video ${index + 1} of ${realSlideCount}`
            );
          } else {
            bullet.removeAttribute(
              "aria-current"
            );

            bullet.setAttribute(
              "aria-label",
              `Go to video ${index + 1} of ${realSlideCount}`
            );
          }
        });
      };

      /*
       * CREATE PAGINATION BUTTONS
       */
      for (
        let index = 0;
        index < realSlideCount;
        index++
      ) {
        const bullet =
          document.createElement("button");

        bullet.type = "button";

        bullet.className =
          "swiper-pagination-bullet";

        bullet.setAttribute(
          "aria-label",
          `Go to video ${index + 1} of ${realSlideCount}`
        );

        bullet.addEventListener(
          "click",
          () => {
            /*
             * Remember which video the user
             * explicitly selected.
             */
            selectedIndex = index;

            /*
             * Move Swiper as far as it can.
             */
            swiper.slideTo(index);

            /*
             * Keep the explicitly selected
             * pagination dot active.
             */
            updatePagination(index);
          }
        );

        const listItem =
          document.createElement("li");

        listItem.appendChild(bullet);
        pagination.appendChild(listItem);

        bullets.push(bullet);
      }

      /*
       * UPDATE PAGINATION WHEN A SLIDE
       * RECEIVES KEYBOARD FOCUS
       */
      realSlides.forEach((slide, index) => {
        const focusable = slide.querySelector(
          "[data-video-open]"
        );
      
        focusable?.addEventListener("focus", () => {
          selectedIndex = index;
          updatePagination(index);
        });
      });

      /*
       * Swipe / keyboard navigation updates
       * pagination from Swiper's position.
       */
      swiper.on(
        "slideChange",
        () => {
          selectedIndex = swiper.activeIndex;
          updatePagination();
        }
      );

      updatePagination();
    }
