	const header = document.querySelector('#top-nav');
	const mainContent = document.querySelector('#main-content');

	if (!header || !mainContent) return;

	const leadSection =
		mainContent.querySelector('.section-under-header, .section-under-header--strong') ||
		mainContent.querySelector('section.cover-item, section.hero');

	if (!leadSection) return;

	function clamp(value, min, max) {
		return Math.min(Math.max(value, min), max);
	}

	function lerp(start, end, progress) {
		return start + (end - start) * progress;
	}

	function getLeadSectionMetrics() {
		const rect = leadSection.getBoundingClientRect();

		return {
			top: window.scrollY + rect.top,
			height: leadSection.offsetHeight || rect.height
		};
	}

	function updateFallbackFadeClass() {
		const hasExplicitFade =
			leadSection.classList.contains('section-under-header') ||
			leadSection.classList.contains('section-under-header--strong');

		mainContent.classList.remove('has-under-header-fade', 'has-under-header-fade-strong');

		if (!hasExplicitFade) {
			/* legacy fallback */
			mainContent.classList.add('has-under-header-fade');

			if (leadSection.matches('.cover-item.is-bright, .hero.is-bright')) {
				mainContent.classList.remove('has-under-header-fade');
				mainContent.classList.add('has-under-header-fade-strong');
			}
		}
	}

	function updateHeaderState() {
		const metrics = getLeadSectionMetrics();
		if (!metrics.height) return;

		const progress = clamp((window.scrollY - metrics.top) / metrics.height, 0, 1);

		header.style.setProperty('--header-progress', progress.toFixed(3));
		header.style.setProperty('--header-bg-alpha', lerp(0, 0.96, progress).toFixed(3));
		header.style.setProperty('--header-blur', `${lerp(0, 10, clamp(progress / 0.6, 0, 1)).toFixed(1)}px`);
		header.style.setProperty('--header-shadow-alpha', lerp(0, 0.08, progress).toFixed(3));

		header.classList.toggle('header-dark-text', progress >= 0.55);
		header.classList.toggle('header-light-text', progress < 0.55);
	}

	updateFallbackFadeClass();
	updateHeaderState();

	window.addEventListener('scroll', updateHeaderState, { passive: true });
