      switcher.classList.add('is-open');

      const overlay = document.querySelector('.nav-overlay');
      if (overlay) {
        const header = document.querySelector('.header');
        const headerBottom = header ? header.getBoundingClientRect().bottom : 0;
        const menuHeight = menu.offsetHeight;
        overlay.style.setProperty('--submenu-height', (headerBottom + menuHeight + 25) + 'px');
        overlay.classList.add('is-active');
      }
    }

    // ── Mobile detection ──
    const mobileQuery = window.matchMedia('(max-width: 1069px)');

    // ── Mouse: hover to open/close (desktop only, with delay) ──
    var _langHoverTimer = null;
    switcher.addEventListener('mouseenter', () => {
      if (!mobileQuery.matches) {
        clearTimeout(_langHoverTimer);
        _langHoverTimer = setTimeout(() => openMenu(), 150);
      }
    });
    switcher.addEventListener('mouseleave', () => {
      if (!mobileQuery.matches) {
        clearTimeout(_langHoverTimer);
        _langHoverTimer = setTimeout(() => closeMenu(), 150);
      }
    });

    // ── Click: toggle on mobile ──
    trigger.addEventListener('click', function() {
      if (mobileQuery.matches) {
        if (menu.hidden) {
          openMenu();
        } else {
          closeMenu();
        }
      }
    });

    // ── Keyboard: focus opens menu (desktop only) ──
    switcher.addEventListener('focusin', () => { if (!mobileQuery.matches) openMenu(); });

    // ── Keyboard: arrow/enter/space opens and moves focus into menu ──
    trigger.addEventListener('keydown', function(event) {
      if (event.key === 'ArrowDown' || event.key === 'Enter' || event.key === ' ') {
        event.preventDefault();
        openMenu();
        const firstItem = getItems()[0];
        if (firstItem) firstItem.focus();
      }

      if (event.key === 'Escape') {
        event.preventDefault();
        closeMenu(true);
      }
    });

    menu.addEventListener('keydown', function(event) {
      const items = getItems();
      if (!items.length) return;

      const currentIndex = items.indexOf(document.activeElement);

      if (event.key === 'Escape') {
        event.preventDefault();
        closeMenu(true);
      }

      if (event.key === 'ArrowDown') {
        event.preventDefault();
        const nextItem = items[(currentIndex + 1) % items.length];
        if (nextItem) nextItem.focus();
      }

      if (event.key === 'ArrowUp') {
        event.preventDefault();
        const previousItem = items[(currentIndex - 1 + items.length) % items.length];
        if (previousItem) previousItem.focus();
      }

      if (event.key === 'Home') {
        event.preventDefault();
        if (items[0]) items[0].focus();
      }

      if (event.key === 'End') {
        event.preventDefault();
        if (items[items.length - 1]) items[items.length - 1].focus();
      }

      if (event.key === 'Tab' && !event.shiftKey && currentIndex === items.length - 1) {
        closeMenu();
      }

      if (event.key === 'Tab' && event.shiftKey && currentIndex === 0) {
        closeMenu();
      }
    });

    switcher.addEventListener('focusout', function(event) {
      if (!switcher.contains(event.relatedTarget)) {
        closeMenu();
      }
    });
  });
});
