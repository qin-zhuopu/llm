                return order; 
            }
        }
    }

    function initBanner(banner,result){
        var imgHtml = '';
        var _imgHtml = '';
        var controlHtml = '';
        if(banner){
            $('.m-spread').show();
            if(banner.length > 1){
                var controlHtml = '<li class="J_scroll on"></li>';
            }
            for(var i=0;i<banner.length;i++){
                var urlHtml = '';
                if(banner[i].url) urlHtml = 'href="' + banner[i].url;
                imgHtml += '<a '+urlHtml+'" target="_blank" style="display:block;background:#' + banner[i].backColor + ' url(' + banner[i].imagePath + ') no-repeat 50% 0"></a>';
                if(i){
                    controlHtml += '<li class="J_scroll"></li>';
                }
            }

            $('.u-spread').html(imgHtml);
            $('.u-dot ul').html(controlHtml);
            setAutoBottom(0);
            if(banner.length>1){
                setAutoScroll(banner);
                $('.J_spread').hover(function(){
                    clearInterval(oScrollTimer);
                },function(){
                    setAutoScroll(banner);
                });
            }
        }

        if(result){
            $('.m-btSpread').show();
            for(var i=0;i<result.length;i++){
                var _urlHtml = '';
                if(result[i].url) _urlHtml = 'href="' + result[i].url;
                _imgHtml += '<a '+_urlHtml+'" target="_blank" style="background:#' + result[i].backColor + ' url(' + result[i].imagePath + ') no-repeat 50% 0"></a>';
            }

            $('.u-btSpread').html(_imgHtml);
            bannerScrollTo(0);

            $('.u-btSpread a').hover(function(){
                window.clearInterval(_slider);
            },function(){
                setAutoBottom($(this).index());
            })
        }
    }

    function setAutoBottom(index){
        var marginLeft = -130*index;
        _slider =   setInterval(function(){
            var slider = $('.u-btSpread');
            var len = slider.children().length;
            var height = slider.height();
            if(len>1){
                var offset = Math.round(height/len);
                marginLeft -= offset;
                if(marginLeft+height<=0){
                    marginLeft = 0;
                }
                slider.animate({'margin-top':marginLeft},800);
            }
        },4000);
    }

    function setAutoScroll(banner){
        oScrollTimer = setInterval(function() {
            var tNextIndex = $(".topDot .on").index() + 1;
            if (tNextIndex == banner.length) {
                tNextIndex = 0;
            }
            bannerScrollTo(tNextIndex);
        }, 4000);
    }

    function bannerScrollTo(tNextIndex){
        var tCurDom = $(".topDot .on");
        $(".u-spread a").eq(tCurDom.index()).fadeOut();
        $(".u-spread a").eq(tNextIndex).fadeIn();
        tCurDom.removeClass("on");
        $(".u-dot li").eq(tNextIndex).addClass("on");
    }
